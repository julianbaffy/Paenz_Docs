\newcommand{\headersubsection}{}

\ifshowbreadcrumbs

  }
  \renewcommand{\subsectionmark}[1]{%
    \renewcommand{\headersubsection}{\hspace{0.6em}--\hspace{0.6em}#1}%
  }

% Vereinsfarbe: übernommen aus custom.css (--kj-accent)
\definecolor{vereinfarbe}{HTML}{2F5E4E}

\setmainfont{Arial}



% Auszeichnungsschrift für Überschriften: Source Serif 4 — seriöse Antiqua,
% passend zu offiziellen Vereinsdokumenten. Fallback Georgia.
\IfFontExistsTF{Source Serif 4}{
  \newfontfamily\displayfont{Source Serif 4}
}{
  \newfontfamily\displayfont{Georgia}
}

% --- Überschriften: Auszeichnungsschrift, Vereinsfarbe, Trennlinie darunter ---
% (ersetzt sectsty vollständig, da sectsty und titlesec sich in die Quere
%  kommen können, wenn beide dieselben Sectioning-Befehle patchen)
\usepackage{titlesec}
\titleformat{\section}
  {\displayfont\color{vereinfarbe}\Large\bfseries}
  {}{0pt}{}
  [{\color{vereinfarbe}\vspace{0.2cm}\titlerule[0.8pt]}]
\titlespacing*{\section}{0pt}{1.2cm}{0.6cm}


\titleformat{\subsection}
  {\displayfont\color{vereinfarbe}\large\bfseries}
  {}{0pt}{}
\titlespacing*{\subsection}{0pt}{0.8cm}{0.3cm}
